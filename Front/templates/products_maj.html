document.addEventListener("DOMContentLoaded", function () { const cardContainer = document.querySelector(".card-container"); const originalCards = Array.from(document.querySelectorAll(".card")); const cardCount = document.getElementById("card-count"); function updateCardCount(count) { cardCount.textContent = `Найдено: ${count}`; } function applyFilters() { const nameFilter = $("#name").val() || []; const gearboxFilter = $("#gearbox").val() || []; const energyFilter = $("#energy").val() || []; const typeFilter = $("#type").val() || []; const mileageFilter = parseInt(document.getElementById("mileage").value); const priceFilter = parseInt(document.getElementById("price").value); const modelFilter = document.getElementById("model").value.toLowerCase(); const mileageCountChecked = document.getElementById("mileage_count").checked; cardContainer.innerHTML = ""; let visibleCount = 0; originalCards.forEach(card => { const cardName = (card.dataset.name || "").toLowerCase(); const cardGearbox = (card.dataset.gearbox || "").toLowerCase(); const cardEnergy = (card.dataset.energy || "").toLowerCase(); const cardType = (card.dataset.type || "").toLowerCase(); const cardMileage = parseInt(card.dataset.mileage) || 0; const cardPrice = parseInt(card.dataset.openingbid) || 0; const cardModel = (card.dataset.model || "").toLowerCase(); const cardMileageCount = parseInt(card.dataset.mileage_count) || 0; let isVisible = true; // OR filters if (nameFilter.length && !nameFilter.some(v => cardName.includes(v))) isVisible = false; if (gearboxFilter.length) { const match = gearboxFilter.some(v => v === "vide" ? cardGearbox === "" : cardGearbox.includes(v) ); if (!match) isVisible = false; } if (energyFilter.length) { const match = energyFilter.some(v => v === "vide" ? cardEnergy === "" : cardEnergy.includes(v) ); if (!match) isVisible = false; } if (typeFilter.length) { const match = typeFilter.some(v => v === "vide" ? cardType === "" : cardType.includes(v) ); if (!match) isVisible = false; } // Numeric filters if (mileageFilter && cardMileage > mileageFilter) isVisible = false; if (priceFilter && cardPrice > priceFilter) isVisible = false; if (modelFilter && !cardModel.includes(modelFilter)) isVisible = false; if (mileageCountChecked && cardMileageCount <= 0) isVisible = false; if (isVisible) { const colDiv = document.createElement("div"); colDiv.className = card.parentElement.className; colDiv.appendChild(card); cardContainer.appendChild(colDiv); visibleCount++; } }); updateCardCount(visibleCount); } // Init Select2 $("#name, #gearbox, #energy, #type").select2({ allowClear: true, width: "100%" }); $("#name, #gearbox, #energy, #type") .on("select2:select select2:unselect", applyFilters); document.querySelectorAll("#filter-form input") .forEach(el => { el.addEventListener("input", applyFilters); el.addEventListener("change", applyFilters); }); document.querySelector("#filter-form button[type='reset']") .addEventListener("click", function () { $("#name, #gearbox, #energy, #type") .val(null) .trigger("change"); cardContainer.innerHTML = ""; originalCards.forEach(card => { const colDiv = document.createElement("div"); colDiv.className = card.parentElement.className; colDiv.appendChild(card); cardContainer.appendChild(colDiv); }); updateCardCount(originalCards.length); }); // Initial count updateCardCount(originalCards.length); });
